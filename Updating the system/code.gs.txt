// ================= SAFE JSON & LOAD QUESTIONS =================
function safeParseJSON(str) {
  if (!str || typeof str !== "string") return null;
  try { return JSON.parse(str); } catch (e) { return null; }
}

function loadQuestions() {
  if (QUESTIONS_CACHE && Date.now() - CACHE_TIME < CACHE_DURATION) return QUESTIONS_CACHE;
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const data = ss.getSheetByName(SHEETS.QUESTIONS).getDataRange().getValues();
  data.shift();
  const seen = new Set();
  QUESTIONS_CACHE = data.map(r => ({
    qid: String(r[0] || "").trim(),
    type: String(r[1] || "single").trim(),
    text: r[2] || "",
    options: safeParseJSON(r[3]) || [],
    gridRows: safeParseJSON(r[4]) || [],
    gridCols: safeParseJSON(r[5]) || [],
    correct: safeParseJSON(r[6]) || null,
    points: Number(r[7]) || 1,
    image: r[8] || "",
    timeLimit: Number(r[9]) || 60
  })).filter(q => {
    if (seen.has(q.qid)) { console.warn("Duplicate QID:", q.qid); return false; }
    seen.add(q.qid);
    return true;
  });
  CACHE_TIME = Date.now();
  return QUESTIONS_CACHE;
}

function startSession(name, email, afa, track) {
  if (!["Primary", "Secondary"].includes(track)) throw new Error("Invalid track");
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const allowed = ss.getSheetByName(SHEETS.ALLOWED_EMAILS).getDataRange().getValues();
  const user = allowed.find(r => String(r[0]).trim() === name && String(r[1]).trim() === email && String(r[2] || "").trim() === afa);
  if (!user) throw new Error("Invalid credentials");
  if ((ss.getSheetByName(SHEETS.ADMIN).getRange("B2").getValue() || "OPEN") !== "OPEN") throw new Error("Quiz is closed");
  const allowMultiple = String(user[3] || "").toUpperCase() === "YES";
  const sessions = ss.getSheetByName(SHEETS.SESSIONS);
  const finished = sessions.getDataRange().getValues().find(r => r[2] === email && (r[9] === true || r[9] === "TRUE"));
  if (finished && !allowMultiple) throw new Error("Already completed");

  const questions = loadQuestions();
  const normalIndices = questions.map((_, i) => i).filter(i => !["Primary29","Primary30","Secondary29","Secondary30"].includes(questions[i].qid));
  shuffleArray(normalIndices);
  const order = normalIndices.slice(0, 28);

  const sessionId = Utilities.getUuid();
  sessions.appendRow([sessionId, name, email, afa, track, JSON.stringify(order), 0, new Date(), 0, false]);
  return { sessionId, track, totalQuestions: 30 };

// ================= DETAILED GRADING =================
function gradeAnswer(q, selected) {
  if (!q.correct) return { isCorrect: null, pointsEarned: 0, details: "No correct answer defined" };

  const correct = q.correct;
  let pointsEarned = 0;
  let details = {};

  if (q.type === "single") {
    const isCorrect = String(selected) === String(correct);
    pointsEarned = isCorrect ? q.points : 0;
    details = { user: selected, correct: correct, match: isCorrect };

  } else if (q.type === "multiple") {
    const userArr = Array.isArray(selected) ? selected.map(String).sort() : [];
    const correctArr = Array.isArray(correct) ? correct.map(String).sort() : [];
    const isCorrect = JSON.stringify(userArr) === JSON.stringify(correctArr);
    pointsEarned = isCorrect ? q.points : 0;
    details = { user: userArr, correct: correctArr, match: isCorrect };

  } else if (q.type === "checkbox_grid" || q.type === "mcq_grid") {
    const userGrid = selected || {};
    const correctGrid = correct || {};
    let correctCells = 0;
    let totalCells = 0;

    q.gridRows.forEach((_, r) => {
      const userRow = userGrid[r] || (q.type === "mcq_grid" ? null : []);
      const correctRow = correctGrid[r] || (q.type === "mcq_grid" ? null : []);
      if (q.type === "mcq_grid") {
        totalCells++;
        if (String(userRow) === String(correctRow)) correctCells++;
      } else {
        const u = Array.isArray(userRow) ? userRow.map(String).sort() : [];
        const c = Array.isArray(correctRow) ? correctRow.map(String).sort() : [];
        totalCells += c.length;
        correctCells += u.filter(x => c.includes(x)).length;
      }
    });
    pointsEarned = totalCells > 0 ? Math.round((correctCells / totalCells) * q.points * 10) / 10 : 0;
    details = { userGrid, correctGrid, correctCells, totalCells, pointsEarned };
  }

  return { isCorrect: pointsEarned === q.points, pointsEarned, details: JSON.stringify(details) };
}

function submitAnswer(sessionId, payload, userEmail) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(SHEETS.SESSIONS);
  const row = findSessionRow(sheet, sessionId);
  if (!row) throw new Error("Session expired");
  const data = sheet.getRange(row, 1, 1, 10).getValues()[0];
  if (data[2] !== userEmail) throw new Error("Unauthorized");

  const idx = Number(data[6] || 0);
  const questions = loadQuestions();
  let q;
  if (idx < 28) {
    q = questions[JSON.parse(data[5])[idx]];
  } else {
    const target = data[4] === "Primary" ? (idx === 28 ? "Primary29" : "Primary30") : (idx === 28 ? "Secondary29" : "Secondary30");
    q = questions.find(qq => qq.qid === target);
  }

  const grade = gradeAnswer(q, payload.selected);
  const isCorrect = grade.isCorrect;
  const pointsEarned = grade.pointsEarned;
  const details = grade.details;

  ss.getSheetByName(SHEETS.RESPONSES).appendRow([
    new Date(), sessionId, userEmail, data[1], q.qid,
    JSON.stringify(payload.selected), isCorrect, payload.timeTaken || 0, data[8],
    pointsEarned, details
  ]);

  sheet.getRange(row, 7).setValue(idx + 1);
  if (idx + 1 >= 30) sheet.getRange(row, 10).setValue(true);

  return { ok: true, nextIndex: idx + 1 };
}

function getParticipantResult(email) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const adminSheet = ss.getSheetByName(SHEETS.ADMIN);
  if (adminSheet.getRange("B4").getValue() !== "YES") throw new Error("Results not released yet");

  const responses = ss.getSheetByName(SHEETS.RESPONSES).getDataRange().getValues();  // ← FIXED
  const sessions = ss.getSheetByName(SHEETS.SESSIONS).getDataRange().getValues();
  const questions = loadQuestions();

  const sessionRow = sessions.find(r => r[2] === email && (r[9] === true || r[9] === "TRUE"));
  if (!sessionRow) throw new Error("No completed attempt found");

  const participantResponses = responses.filter(r => r[2] === email).sort((a, b) => a[0] - b[0]);

  let totalScore = 0, maxScore = 0;
  const details = [];

  participantResponses.forEach(resp => {
    const q = questions.find(qq => qq.qid === resp[4]);
    if (!q) return;
    const userAnswer = safeParseJSON(resp[5]);
    const gradeDetails = safeParseJSON(resp[10]) || {};
    const pointsEarned = Number(resp[9]) || 0;

    totalScore += pointsEarned;
    maxScore += q.points;

    details.push({
      question: q.text,
      image: q.image || "",
      type: q.type,
      options: q.options,
      gridRows: q.gridRows,
      gridCols: q.gridCols,
      userAnswer: userAnswer,
      correctAnswer: q.correct,
      isCorrect: resp[6] === true,
      pointsEarned: pointsEarned,
      pointsPossible: q.points,
      timeTaken: Math.round(resp[7] || 0),
      feedback: gradeDetails
    });
  });

  return {
    name: sessionRow[1], email, track: sessionRow[4],
    totalScore, maxScore, percentage: Math.round((totalScore / maxScore) * 100),
    details
  };
}

/**
 * FINAL RESULT TAB GENERATOR — 100% ACCURATE
 * Fixes checkbox_grid, Points, QuestionText, Percentage
 * Add this to your existing Code.gs — DO NOT delete anything else!
 */

function generateResultTab() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const resp = ss.getSheetByName("Responses");
  const q = ss.getSheetByName("Questions");
  if (!resp || !q) return SpreadsheetApp.getUi().alert("Missing sheet!");

  const rData = resp.getDataRange().getValues();
  const qData = q.getDataRange().getValues();
  if (rData.length < 2) return;

  const rh = rData[0], qh = qData[0];
  const r = {email:rh.indexOf("Email"), name:rh.indexOf("Name"), qid:rh.indexOf("QID"), sel:rh.indexOf("Selected(JSON)")};
  const qq = {qid:qh.indexOf("QID"), text:qh.indexOf("QuestionText"), opt:qh.indexOf("OptionsJSON"), cor:qh.indexOf("CorrectJSON"), pts:qh.indexOf("Points"), type:qh.indexOf("Type")};

  // Question map — with CLEAN CorrectJSON display
  const qMap = {};
  for (let i = 1; i < qData.length; i++) {
    const row = qData[i];
    const qid = row[qq.qid];
    if (!qid) continue;

    let rawCorrect = row[qq.cor] || "";
    let parsedCorrect = null;

    // Parse once for grading
    try {
      parsedCorrect = rawCorrect ? JSON.parse(rawCorrect.replace(/,\s*}/g,"}").replace(/,\s*]/g,"]")) : null;
    } catch(e) { parsedCorrect = null; }

    // Ensure checkbox_grid has arrays
    if (parsedCorrect && typeof parsedCorrect === "object" && !Array.isArray(parsedCorrect)) {
      Object.keys(parsedCorrect).forEach(k => {
        if (!Array.isArray(parsedCorrect[k])) parsedCorrect[k] = [parsedCorrect[k]];
      });
    }

    qMap[qid] = {
      text: row[qq.text] || "",
      opt: row[qq.opt] || "",
      correctParsed: parsedCorrect,        // for grading
      correctDisplay: rawCorrect || "null", // for display — NO extra escaping
      pts: Number(row[qq.pts]) || 0,
      type: String(row[qq.type] || "single").trim()
    };
  }

  // Group participants
  const participants = {};
  for (let i = 1; i < rData.length; i++) {
    const row = rData[i];
    const email = row[r.email];
    if (!email || row[r.sel] === "VIOLATION") continue;
    if (!participants[email]) participants[email] = {name: row[r.name]||"", rows:[]};
    participants[email].rows.push(row);
  }

  // Create Result sheet
  let sheet = ss.getSheetByName("Result");
  if (!sheet) sheet = ss.insertSheet("Result"); else sheet.clear();

  const headers = ["Email","Name","QID","Question Text","OptionsJSON","Selected","CorrectJSON","Points","Points Earned","Percentage"];
  sheet.getRange(1,1,1,headers.length).setValues([headers]).setFontWeight("bold").setBackground("#e8f4f8");

  let row = 2;
  Object.keys(participants).sort().forEach(email => {
    const p = participants[email];
    let totalPts = 0, totalEarned = 0;

    p.rows.forEach(respRow => {
      const qid = respRow[r.qid]; 
      const selRaw = respRow[r.sel];
      const qObj = qMap[qid];
      if (!qObj) { row++; return; }

      let sel; try { sel = JSON.parse(selRaw); } catch(e) { sel = selRaw; }

      let correctCnt = 0, totalCells = 0;
      const cor = qObj.correctParsed;

      if (qObj.type === "single") {
        correctCnt = (String(sel) === String(cor)) ? 1 : 0; totalCells = 1;
      }
      else if (qObj.type === "multiple" || qObj.type === "check") {
        const c = Array.isArray(cor) ? cor.map(String) : [];
        const u = Array.isArray(sel) ? sel.map(String) : [];
        correctCnt = u.filter(x => c.includes(x)).length; totalCells = c.length;
      }
      else if (qObj.type === "mcq_grid") {
        Object.keys(cor||{}).forEach(k => {
          if (String(sel?.[k]) === String(cor[k])) correctCnt++;
        });
        totalCells = Object.keys(cor||{}).length;
      }
      // ──────── CHECKBOX GRID – TRUE CELL-BY-CELL COMPARISON (FIXED) ────────
      // CHECKBOX GRID – STRICT ROW-BY-ROW EXACT MATCH (2 points per perfect row)
      else if (qObj.type === "checkbox_grid") {
        let perfectRows = 0;
        let totalRows   = 0;

        Object.keys(cor || {}).forEach(rowKey => {
          const correctOptions = Array.isArray(cor[rowKey]) ? cor[rowKey].map(String).sort() : [];
          const userOptions    = Array.isArray(sel?.[rowKey]) ? sel[rowKey].map(String).sort() : [];

          totalRows++;

          // Exact match: same options, same length, no extra/no missing
          if (correctOptions.length === userOptions.length &&
              correctOptions.every(opt => userOptions.includes(opt))) {
            perfectRows++;
          }
        });

        correctCnt = perfectRows;     // 1 "correct count" = 1 perfect row
        totalCells = totalRows;       // total possible perfect rows
      }

      const earned = correctCnt * 2;
      totalPts += qObj.pts;
      totalEarned += earned;

      sheet.getRange(row++,1,1,headers.length).setValues([[
        email, p.name, qid, qObj.text, qObj.opt, selRaw,
        qObj.correctDisplay,    // CLEAN — no backslashes!
        qObj.pts, earned,
        totalCells ? Math.round((correctCnt/totalCells)*100)+"%" : "0%"
      ]]);
    });

function createBeautifulResultHTML(r) {
  const rows = r.details.map((q, i) => {
    let answerHTML = "<em>No response</em>";

    if (q.type === "single" || q.type === "multiple" || q.type === "check") {
      const correct = Array.isArray(q.correctAnswer) ? q.correctAnswer.map(String) : [String(q.correctAnswer || "")];
      const user    = q.userAnswer ? (Array.isArray(q.userAnswer) ? q.userAnswer.map(String) : [String(q.userAnswer)]) : [];

      answerHTML = (q.options || []).map((opt, idx) => {
        const isCorrect = correct.includes(String(idx));
        const isSelected = user.includes(String(idx));
        let style = "padding:7px 12px; margin:4px; border-radius:8px; font-size:0.95em; display:inline-block;";
        if (isCorrect && isSelected)  style += "background:#d4edda; color:#166534; border:2px solid #22c55e;";
        else if (isCorrect)           style += "background:#86efac; color:#166534;";
        else if (isSelected)          style += "background:#fecaca; color:#991b1b; border:2px solid #ef4444;";
        else                          style += "background:#f1f5f9; color:#64748b;";
        return `<span style="${style}">${opt}</span>`;
      }).join("");

      (q.gridRows || []).forEach((row, rIdx) => {
        table += `<tr><td style="padding:8px; font-weight:bold; width:30%;">${row}</td><td style="padding:8px;">`;

          const key = String(rIdx);

        });

// ──────────────────────────────────────────────────────────────
// QUESTION-WISE ANALYTICS (Overall + AFA-wise % correct)
// ──────────────────────────────────────────────────────────────
function generateQuestionAnalyticsHTML() {
  const MAIN_RESULT_FOLDER_ID = "1SHZRUqX-gtNdg76rI3TMMyii4ANK_Jl5"; // ← same as before
  const folder = DriveApp.getFolderById(MAIN_RESULT_FOLDER_ID);

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const adminSheet = ss.getSheetByName("Admin");
  if (adminSheet.getRange("B4").getValue() !== "YES") {
    SpreadsheetApp.getUi().alert("Results not released yet!");
    return;
  }

  const allowed = ss.getSheetByName("Allowed_Emails").getDataRange().getValues();
  const pins    = ss.getSheetByName("PINs").getDataRange().getValues();

  // Build email → AFA map
  const emailToAFA = {};
  for (let i = 1; i < allowed.length; i++) {
    const email = String(allowed[i][1] || "").trim().toLowerCase();
    const afa   = String(allowed[i][2] || "Others").trim();
    if (email) emailToAFA[email] = afa;
  }

  // Collect all results
  const allResults = [];
  for (let i = 1; i < pins.length; i++) {
    const email = String(pins[i][0] || "").trim().toLowerCase();
    const pin   = String(pins[i][2] || "").trim().toUpperCase();
    if (!email) continue;
    try {
      const r = validatePinAndGetResult(email, pin);
      r.afa = emailToAFA[email] || "Others";
      allResults.push(r);
    } catch (e) { /* skip invalid */ }
  }

  if (allResults.length === 0) {
    SpreadsheetApp.getUi().alert("No results found!");
    return;
  }

  // Question stats: qid → {text, overallCorrect, total, afa:{AFA: {correct, total}}}
  const qStats = {};
  allResults.forEach(r => {
    r.details.forEach(q => {
      if (!qStats[q.qid]) {
        qStats[q.qid] = {
          text: q.question.replace(/<[^>]*>/g, "").substring(0, 100) + (q.question.length > 100 ? "…" : ""),
          overall: { correct: 0, total: allResults.length },
          afa: {}
        };
      }
      const isCorrect = q.pointsEarned === q.pointsPossible;
      if (isCorrect) qStats[q.qid].overall.correct++;

      const afa = r.afa;
      if (!qStats[q.qid].afa[afa]) qStats[q.qid].afa[afa] = { correct: 0, total: 0 };
      qStats[q.qid].afa[afa].total++;
      if (isCorrect) qStats[q.qid].afa[afa].correct++;
    });
  });

  // Build HTML rows
  let rows = "";
  Object.keys(qStats).sort((a, b) => {
    const pa = qStats[a].overall.correct / qStats[a].overall.total;
    const pb = qStats[b].overall.correct / qStats[b].overall.total;
    return pa - pb; // hardest first
  }).forEach(qid => {
    const q = qStats[qid];
    const overallPct = Math.round((q.overall.correct / q.overall.total) * 100);

    let afaCells = "";
    Object.keys(q.afa).sort().forEach(afa => {
      const p = Math.round((q.afa[afa].correct / q.afa[afa].total) * 100);
      const color = p >= 70 ? "#166534" : p >= 40 ? "#ca8a04" : "#991b1b";
      afaCells += `<td style="background:${p>=70?'#ecfdf5':p>=40?'#fffbeb':'#fee2e2'};color:${color};font-weight:600;">${p}%</td>`;
    });

    rows += `
      <tr style="background:${overallPct>=70?'#f0fdf4':overallPct>=40?'#fffbeb':'#fef2f2'};">
        <td style="font-weight:600;">${qid}</td>
        <td>${q.text}</td>
        <td style="font-weight:bold;font-size:1.1em;color:${overallPct>=70?'#166534':overallPct>=40?'#ca8a04':'#991b1b'};">${overallPct}%</td>
        ${afaCells}
      </tr>`;
  });

  // AFA header
  const afaHeader = Object.keys(qStats[Object.keys(qStats)[0]]?.afa || {}).sort().map(a => `<th>${a}</th>`).join("");

  const file = folder.createFile("Question_Analytics.html", html, MimeType.HTML);
